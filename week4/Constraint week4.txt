def safe(n, c, a, g):   
    return all(a.get(i) != c for i in g[n])
def bt(g, col, a):
    if len(a) == len(g):
        return a
    n = next(i for i in g if i not in a)
    for c in col:
        if safe(n, c, a, g):
            a[n] = c
            r = bt(g, col, a)
            if r:
                return r
            a.pop(n)
    return None
nodes = input("Nodes: ").split()
g = {n: input(f"Neighbors of {n}: ").split() for n in nodes}
colors = input("Colors: ").split()
s = bt(g, colors, {})
print("Assigned Colors:")
print("\n".join(f"{k} : {v}" for k, v in s.items()) if s else "No solution")

Neighbors of D: B C
