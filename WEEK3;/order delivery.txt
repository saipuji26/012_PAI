import heapq
pos = {1:(0,0),2:(0,1),3:(0,2),4:(1,0),5:(1,1),6:(1,2),7:(2,0),8:(2,1),9:(2,2)}
start = int(input("Start house: "))
goal = int(input("Goal house: "))
h = lambda p: abs(p[0]-pos[goal][0]) + abs(p[1]-pos[goal][1])
pq = [(h(pos[start]), 0, pos[start])]
par = {pos[start]: None}
g = {pos[start]: 0}
while pq:
    _, c, p = heapq.heappop(pq)
    if p == pos[goal]:
        path = []
        while p:
            path.append(p)
            p = par[p]
        for i, s in enumerate(path[::-1], 1):
            house = [k for k,v in pos.items() if v==s][0]
            print(f"Step {i}: House {house} -> {s}")
        break
    for dx, dy in [(1,0),(0,1),(-1,0),(0,-1)]:
        np = (p[0]+dx, p[1]+dy)
        if 0 <= np[0] < 3 and 0 <= np[1] < 3:
            nc = c + 1
            if np not in g or nc < g[np]:
                g[np] = nc
                par[np] = p
                heapq.heappush(pq, (nc + h(np), nc, np))
