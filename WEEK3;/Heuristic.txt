import heapq
goal = [1,2,3,4,5,6,7,8,0]
def h(s): return sum(abs(i//3 - v//3) + abs(i%3 - v%3) for i,v in enumerate(s) if v)
def moves(s):
    i = s.index(0)
    for d in [-3,3,-1,1]:
        j = i + d
        if 0 <= j < 9 and not (i%3==0 and d==-1) and not (i%3==2 and d==1):
            t = s[:]
            t[i], t[j] = t[j], t[i]
            yield t
def solve(start):
    pq = [(h(start), start, [start])]
    seen = set()
    while pq:
        _, s, path = heapq.heappop(pq)
        if s == goal: return path
        if tuple(s) in seen: continue
        seen.add(tuple(s))
        for n in moves(s):
            heapq.heappush(pq, (len(path)+h(n), n, path+[n]))
start = list(map(int, input("Enter puzzle (0 for blank): ").split()))
res = solve(start)
for step in res:
    for i in range(0,9,3):
        print(step[i:i+3])
    print()
