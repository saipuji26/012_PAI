from hmmlearn import hmm
import numpy as np

np.random.seed(42)

model = hmm.CategoricalHMM(n_components=3, init_params="")

# Slightly favor starting in state 0
model.startprob_ = np.array([0.6, 0.3, 0.1])

# Force transition towards state 1 after step 1
model.transmat_ = np.array([
    [0.2, 0.7, 0.1],   # strongly pushes 0 → 1
    [0.2, 0.6, 0.2],   # stay in 1
    [0.2, 0.3, 0.5]
])

# Make emissions clearly favor switching
model.emissionprob_ = np.array([
    [0.9, 0.1],   # state 0 → strong for observation 0
    [0.2, 0.8],   # state 1 → strong for observation 1
    [0.3, 0.7]
])

observations = np.array([[0], [1], [1], [0]])

logprob, hidden_states = model.decode(observations, algorithm="viterbi")

print("Predicted Hidden Engagement States:", hidden_states)



Output:-
Predicted Hidden Engagement States: [0 1 1 0]
